    Κάρτα: Ανακάτεμα¶

    Με μια ματιά

    • Τι προσθέτει: μια λίστα που ανακατεύεται με το random.shuffle. Στο πρότυπο ο στόχος της κάρτας Συγκρούσεις επισκέπτεται τις θέσεις του με τυχαία σειρά, και ξαναανακατεύεται όταν περάσουν όλες.
    • Αλλάζει: logic.py, game.py και check_logic.py.
    • Χρόνος: περίπου 25 λεπτά.
    • Χρειάζεσαι: την κάρτα Συγκρούσεις και το module random του Μαθήματος 1.

    Πότε τη θέλεις¶

    Όταν έχεις μια λίστα από πράγματα και θέλεις να τα δεις όλα, μία φορά το καθένα, με τυχαία σειρά: ερωτήσεις ενός κουίζ, κάρτες μιας τράπουλας, οι θέσεις ενός στόχου, οι εχθροί ενός κύματος. Με τη random.choice κάθε φορά θα μπορούσες να πάρεις το ίδιο πράγμα δύο φορές στη σειρά· το ανακάτεμα δεν το κάνει.

    Πώς δουλεύει¶

    Αν έκανες την πρόκληση των ανακατεμένων επιλογών του Μαθήματος 4, τη random.shuffle την έχεις ήδη δει. Εδώ τη μαθαίνουμε από την αρχή.

    Η random.shuffle(λίστα) αλλάζει τη λίστα στη θέση της και δεν επιστρέφει τίποτα (επιστρέφει None). Από αυτό βγαίνουν τρία πράγματα που πρέπει να προσέξεις:

    1. Χρειάζεται λίστα. Μια πλειάδα δεν αλλάζει, άρα η random.shuffle της δίνει σφάλμα.
    2. Αν θέλεις να κρατήσεις την αρχική σειρά, ανακάτευε ένα αντίγραφο: list(πλειάδα).
    3. Δεν γράφεις x = random.shuffle(...): το x θα γίνει None.

    Στην κάρτα Συγκρούσεις ο στόχος περνά από τις θέσεις TARGET_SPOTS πάντα με την ίδια σειρά, και ο παίκτης τη μαθαίνει. Εδώ το παιχνίδι φτιάχνει σε κάθε νέο παιχνίδι ένα ανακατεμένο αντίγραφο των θέσεων και το βάζει στα δεδομένα του (game["spots"]). Όταν ο στόχος περάσει από την τελευταία θέση, η λίστα ξαναανακατεύεται.

    Το τυχαίο δεν δοκιμάζεται, γιατί κάθε φορά βγαίνει άλλο αποτέλεσμα. Δοκιμάζεις ό,τι πρέπει να ισχύει όποια κι αν είναι η τυχαία σειρά: ότι οι θέσεις που ανακατεύτηκαν είναι ακριβώς οι ίδιες θέσεις με άλλη σειρά. Αυτό κάνει η is_permutation.

    Τα μπλοκ κώδικα διαβάζονται όπως εξηγείται στην αρχική σελίδα των καρτών. Η κάρτα εφαρμόζεται πάνω σε ένα πρότυπο που έχει ήδη την κάρτα Συγκρούσεις.

    Βήμα 1: το logic.py¶

    Το new_game παίρνει τη λίστα των θέσεων, ήδη ανακατεμένη:

    logic.py · αντικατάσταση: new_game
    def new_game(x, y, spots):                                          # ← ΑΛΛΑΞΕ: νέο όρισμα
    """Τα δεδομένα ενός νέου παιχνιδιού: ο παίκτης στη θέση (x, y), ο χρόνος και οι πόντοι στο μηδέν, και ο στόχος στην πρώτη από τις θέσεις spots."""
    return {"x": x, "y": y, "time": 0.0, "score": 0, "target": 0, "spots": spots}        # ← ΝΕΟ: τα "spots"
    logic.py · προσθήκη στο τέλος
    def is_permutation(first, second):
    """True αν οι λίστες first και second έχουν ακριβώς τα ίδια στοιχεία, με οποιαδήποτε σειρά."""
    return sorted(first) == sorted(second)

    Βήμα 3: το game.py¶

    Η random μπαίνει στις εισαγωγές, στην αρχή του αρχείου:

    game.py · εισαγωγή στην αρχή
    import random

    Και η is_permutation στην εισαγωγή από το logic.py:

    game.py · εισαγωγές από το logic
    from logic import GAME_OVER, MENU, PLAYING, boxes_overlap, is_permutation, move, new_game, next_index, next_state, shrink_box

    Η νέα συνάρτηση φτιάχνει το ανακατεμένο αντίγραφο:

    game.py · κανόνες
    def shuffled_spots():
    """Οι θέσεις του στόχου με τυχαία σειρά: ένα αντίγραφο της TARGET_SPOTS, ανακατεμένο."""
    spots = list(TARGET_SPOTS)                      # αντίγραφο: μια πλειάδα δεν ανακατεύεται, και η TARGET_SPOTS πρέπει να μείνει όπως είναι
    random.shuffle(spots)                           # αλλάζει τη λίστα στη θέση της και δεν επιστρέφει τίποτα
    assert is_permutation(spots, TARGET_SPOTS)      # οι ίδιες θέσεις, με άλλη σειρά
    return spots

    Η update_game παίρνει τη θέση του στόχου από τη λίστα του παιχνιδιού και, όταν ο στόχος γυρίσει στην πρώτη θέση, ανακατεύει ξανά:

    game.py · αντικατάσταση: update_game
    def update_game(game, dt):
    """Ένα καρέ του παιχνιδιού. ΕΔΩ ΜΠΑΙΝΕΙ Ο ΚΑΝΟΝΑΣ ΣΑΣ.

    Αλλάζει τα δεδομένα του game και επιστρέφει μια ενέργεια για τον πίνακα μεταβάσεων (π.χ. «lose» όταν χάνει ο παίκτης),
    ή None αν το παιχνίδι συνεχίζεται."""
    dx, dy = read_direction()
    game["x"] = move(game["x"], dx, PLAYER_SPEED, dt, PLAY_AREA.left, PLAY_AREA.right - PLAYER_SIZE)
    game["y"] = move(game["y"], dy, PLAYER_SPEED, dt, PLAY_AREA.top, PLAY_AREA.bottom - PLAYER_SIZE)
    game["time"] += dt
    player_box = shrink_box((game["x"], game["y"], PLAYER_SIZE, PLAYER_SIZE), HIT_MARGIN)
    spot = game["spots"][game["target"]]                                                              # ← ΑΛΛΑΞΕ: από τη λίστα του παιχνιδιού
    if boxes_overlap(player_box, (spot[0], spot[1], TARGET_SIZE, TARGET_SIZE)):
        game["score"] += 10
        game["target"] = next_index(game["target"], len(game["spots"]))                               # ← ΑΛΛΑΞΕ: len(game["spots"])
        if game["target"] == 0:                                                                       # ← ΝΕΟ: πέρασαν όλες οι θέσεις
            random.shuffle(game["spots"])                                                             # ← ΝΕΟ: νέα σειρά
    if game["time"] >= DEMO_SECONDS:        # ΣΒΗΣΤΕ ΤΟ: ο προσωρινός κανόνας
        return "lose"
    return None

    Η σχεδίαση του στόχου διαβάζει τη θέση από τη λίστα του παιχνιδιού:

    game.py · αντικατάσταση: draw_target
    def draw_target(surface, game):
    """Ζωγραφίζει τον ενεργό στόχο."""
    spot = game["spots"][game["target"]]                                                              # ← ΑΛΛΑΞΕ: από τη λίστα του παιχνιδιού
    pygame.draw.rect(surface, YELLOW, (spot[0], spot[1], TARGET_SIZE, TARGET_SIZE))

    Ο βρόχος φτιάχνει ανακατεμένες θέσεις σε κάθε νέο παιχνίδι:

    game.py · ο βρόχος του παιχνιδιού
    game = new_game(START_X, START_Y, shuffled_spots())                      # ← ΑΛΛΑΞΕ: τρίτο όρισμα
state = MENU

running = True
while running:
    dt = min(clock.tick(FPS) / 1000, MAX_DT)

    for action in handle_events(state):
        new_state = next_state(state, action)
        if action == "quit":
            running = False
        elif new_state != state:
            if action == "play":
                game = new_game(START_X, START_Y, shuffled_spots())      # ← ΑΛΛΑΞΕ: τρίτο όρισμα
            state = new_state

    if state == PLAYING:
        action = update_game(game, dt)
        if action is not None:
            state = next_state(state, action)

    screen.fill(BACKGROUND)
    if state == MENU:
        draw_menu(screen)
    else:
        draw_target(screen, game)
        draw_player(screen, game)
        draw_hud(screen, font, game)
        if state == GAME_OVER:
            draw_game_over(screen, game)
    pygame.display.flip()

pygame.quit()

    Έτοιμο όταν¶

    • Σε κάθε νέο παιχνίδι ο στόχος ξεκινά από άλλη θέση και τις επισκέπτεται με άλλη σειρά.
    • Σε έναν γύρο ο στόχος περνά από όλες τις θέσεις, χωρίς να επαναλαμβάνεται καμία.
    • Όταν περάσουν και οι τέσσερις θέσεις, η σειρά ανακατεύεται ξανά.
    • Η λίστα TARGET_SPOTS μένει όπως ήταν.
    • Περνούν όλες οι δοκιμές του check_logic.py.

    Συνηθισμένα λάθη¶

    Τι βλέπεις Πιθανή αιτία
    TypeError: 'tuple' object does not support item assignment Ανακάτεψες κατευθείαν την πλειάδα TARGET_SPOTS: έγραψες random.shuffle(TARGET_SPOTS) και όχι random.shuffle(spots)
    TypeError: 'NoneType' object is not iterable Έγραψες spots = random.shuffle(list(TARGET_SPOTS)): η random.shuffle δεν επιστρέφει τη λίστα, άρα η spots έγινε None

    Δοκίμασε και κάτι δικό σου¶

    • Πρόσθεσε μια πέμπτη θέση στην TARGET_SPOTS και δες ότι δεν χρειάζεται να αλλάξεις τίποτα άλλο.
    • Τύπωσε τη λίστα spots μετά από κάθε ανακάτεμα, με print, και δες ότι αλλάζει.
    • Πώς θα έκανες το ίδιο με τη random.choice και τη list.remove (Μάθημα 3), χωρίς τη random.shuffle;
